# Required for YouTube. YouTube makes clients solve a JavaScript challenge
# (the "n challenge") before releasing media URLs; without it downloads fail
# with "Requested format is not available" or "HTTP Error 403: Forbidden".
# This package ships the solver locally, so --remote-components is not needed.
#
# IMPORTANT: a JavaScript runtime must also be installed, and it has to meet
# yt-dlp's minimum version. Node.js 18 is NOT enough.
#     Node.js 22+    https://nodejs.org
#     Deno 2.3+      https://deno.com
#     Bun 1.2.11+    https://bun.sh
# The app probes the runtime on startup and reports it as missing or too old.
# Transcribing local files works fine without one.
yt-dlp-ejs>=0.8.0

# GPU acceleration (NVIDIA).
# CTranslate2 needs cuBLAS 12 and cuDNN 9. These wheels ship the DLLs, so the
# CUDA Toolkit does NOT have to be installed. On machines without an NVIDIA GPU
# you can drop these two lines — the app falls back to CPU automatically.
nvidia-cublas-cu12; platform_system != "Darwin"
nvidia-cudnn-cu12>=9,<10; platform_system != "Darwin"
